# Linux editor toolkit prototypes (docs/LinuxEditorPlan.md, Change 1).
# Standalone measurement project: it is never part of the LayerC build and
# adds no toolkit dependency to any shipped target.
#   cmake -S tools/editor-prototypes -B <build> -G Ninja -DCMAKE_BUILD_TYPE=Release
cmake_minimum_required(VERSION 3.28)
project(LayerCEditorPrototypes LANGUAGES C CXX)

set(LAYERC_BUILD_DAEMON OFF CACHE BOOL "" FORCE)
set(LAYERC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LAYERC_BUILD_EDITOR ON CACHE BOOL "" FORCE)
add_subdirectory(../.. layerc EXCLUDE_FROM_ALL)

find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 IMPORTED_TARGET "gtk4>=4.12")
# Older adaptors have a cache weak-reference cleanup defect (GNOME #213/#231).
pkg_check_modules(ATK_BRIDGE QUIET "atk>=2.62" "atk-bridge-2.0>=2.62" "atspi-2>=2.62")
# fltk-config is the portable FLTK 1.4 query; LAYERC_PROTO_SYSROOT relocates
# its /usr paths when the packages were extracted without installation.
find_program(FLTK_CONFIG NAMES "${CMAKE_LIBRARY_ARCHITECTURE}-fltk-config" fltk-config)
set(LAYERC_PROTO_SYSROOT "" CACHE PATH "Prefix that replaces /usr in fltk-config output")

# Bind the version-checked bridge metadata to the same installation's headers
# and libraries. Bridge-owned components (ATK, its AT-SPI adaptor and
# libatspi) must come from that installation or configuration fails: a
# same-named system copy could be the older, defective adaptor. Shared
# dependencies such as GLib and D-Bus may resolve from the system.
set(ATK_BRIDGE_COMPONENT_LIBRARIES atk-1.0 atk-bridge-2.0 atspi)
set(ATK_BRIDGE_COMPONENT_HEADERS "/atk-1.0$" "/at-spi2-atk/2.0$" "/at-spi-2.0$")
if(ATK_BRIDGE_FOUND)
    # The version-checked metadata must itself come from the selected
    # installation: with a relocated prefix, every component's .pc file must
    # live in that prefix; otherwise in its own declared libdir.
    # Each component library is then resolved only from its own module's
    # (relocated) libdir, so metadata and artifact share one installation.
    foreach(module IN ITEMS atk atk-bridge-2.0 atspi-2)
        pkg_get_variable(module_pc_dir "${module}" pcfiledir)
        pkg_get_variable(module_libdir "${module}" libdir)
        file(REAL_PATH "${module_pc_dir}" module_pc_dir)
        if(LAYERC_PROTO_SYSROOT)
            file(REAL_PATH "${LAYERC_PROTO_SYSROOT}" sysroot_dir)
            cmake_path(IS_PREFIX sysroot_dir "${module_pc_dir}" NORMALIZE metadata_in_sysroot)
            if(NOT metadata_in_sysroot)
                message(FATAL_ERROR "${module} metadata ${module_pc_dir} is not from ${LAYERC_PROTO_SYSROOT}")
            endif()
            set(module_libdir "${LAYERC_PROTO_SYSROOT}${module_libdir}")
        else()
            file(REAL_PATH "${module_libdir}/pkgconfig" declared_pc_dir)
            if(NOT module_pc_dir STREQUAL declared_pc_dir)
                message(FATAL_ERROR "${module} metadata ${module_pc_dir} does not match its libdir ${module_libdir}")
            endif()
        endif()
        file(REAL_PATH "${module_libdir}" "ATK_BRIDGE_LIBDIR_${module}")
    endforeach()
    set(ATK_BRIDGE_MODULE_atk-1.0 atk)
    set(ATK_BRIDGE_MODULE_atk-bridge-2.0 atk-bridge-2.0)
    set(ATK_BRIDGE_MODULE_atspi atspi-2)
    set(ATK_BRIDGE_LINK)
    set(ATK_BRIDGE_RUNTIME_DIRS)
    set(ATK_BRIDGE_INCLUDES)
    foreach(dir IN LISTS ATK_BRIDGE_INCLUDE_DIRS)
        set(component FALSE)
        foreach(pattern IN LISTS ATK_BRIDGE_COMPONENT_HEADERS)
            if(dir MATCHES "${pattern}")
                set(component TRUE)
            endif()
        endforeach()
        if(LAYERC_PROTO_SYSROOT AND EXISTS "${LAYERC_PROTO_SYSROOT}${dir}")
            list(APPEND ATK_BRIDGE_INCLUDES "${LAYERC_PROTO_SYSROOT}${dir}")
        elseif(component AND LAYERC_PROTO_SYSROOT)
            message(FATAL_ERROR "Bridge headers ${dir} are missing from ${LAYERC_PROTO_SYSROOT}")
        elseif(EXISTS "${dir}")
            list(APPEND ATK_BRIDGE_INCLUDES "${dir}")
        elseif(component)
            message(FATAL_ERROR "Bridge headers ${dir} named by pkg-config do not exist")
        endif()
    endforeach()
    set(atk_search_dirs ${ATK_BRIDGE_LIBRARY_DIRS})
    if(NOT atk_search_dirs)
        set(atk_search_dirs "/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}" /usr/lib)
    endif()
    foreach(library IN LISTS ATK_BRIDGE_LIBRARIES)
        list(FIND ATK_BRIDGE_COMPONENT_LIBRARIES "${library}" component)
        set(hints)
        if(component GREATER_EQUAL 0)
            set(owning_libdir "${ATK_BRIDGE_LIBDIR_${ATK_BRIDGE_MODULE_${library}}}")
            unset(atk_library CACHE)
            find_library(atk_library NAMES "${library}" HINTS "${owning_libdir}" NO_DEFAULT_PATH)
            if(NOT atk_library)
                message(FATAL_ERROR "Bridge library ${library} is missing from its module libdir ${owning_libdir}")
            endif()
            file(REAL_PATH "${atk_library}" canonical_library)
            get_filename_component(canonical_library_dir "${canonical_library}" DIRECTORY)
            if(NOT canonical_library_dir STREQUAL owning_libdir)
                message(FATAL_ERROR "Bridge library ${library} resolves outside ${owning_libdir}")
            endif()
            list(APPEND ATK_BRIDGE_LINK "${atk_library}")
            list(APPEND ATK_BRIDGE_RUNTIME_DIRS "${owning_libdir}")
            unset(atk_library CACHE)
            continue()
        endif()
        foreach(dir IN LISTS atk_search_dirs)
            if(LAYERC_PROTO_SYSROOT AND EXISTS "${LAYERC_PROTO_SYSROOT}${dir}/lib${library}.so")
                list(APPEND hints "${LAYERC_PROTO_SYSROOT}${dir}")
            endif()
        endforeach()
        if(NOT hints)
            if(component GREATER_EQUAL 0 AND LAYERC_PROTO_SYSROOT)
                message(FATAL_ERROR "Bridge library ${library} is missing from ${LAYERC_PROTO_SYSROOT}")
            endif()
            set(hints ${atk_search_dirs})
        endif()
        unset(atk_library CACHE)
        find_library(atk_library NAMES "${library}" HINTS ${hints} NO_DEFAULT_PATH)
        if(NOT atk_library AND component LESS 0)
            # A shared dependency without a development link (e.g. dbus-1) is
            # still resolved by the loader through libatspi's own dependency.
            message(STATUS "Bridge dependency ${library} is linked transitively")
            continue()
        endif()
        if(NOT atk_library)
            message(FATAL_ERROR "Version-checked bridge library ${library} not found in ${hints}")
        endif()
        list(APPEND ATK_BRIDGE_LINK "${atk_library}")
        if(component GREATER_EQUAL 0)
            get_filename_component(atk_dir "${atk_library}" DIRECTORY)
            list(APPEND ATK_BRIDGE_RUNTIME_DIRS "${atk_dir}")
        endif()
        unset(atk_library CACHE)
    endforeach()
    list(REMOVE_DUPLICATES ATK_BRIDGE_RUNTIME_DIRS)
    list(LENGTH ATK_BRIDGE_RUNTIME_DIRS runtime_dir_count)
    if(NOT runtime_dir_count EQUAL 1)
        message(FATAL_ERROR "Bridge libraries come from several installations: ${ATK_BRIDGE_RUNTIME_DIRS}")
    endif()
    # The runtime check binds the loaded copies to this canonical directory.
    file(REAL_PATH "${ATK_BRIDGE_RUNTIME_DIRS}" ATK_BRIDGE_CANONICAL_DIR)
endif()
if(FLTK_CONFIG)
    execute_process(COMMAND "${FLTK_CONFIG}" --version OUTPUT_VARIABLE FLTK_VERSION
        OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(FLTK_VERSION VERSION_LESS 1.4)
        message(FATAL_ERROR "The representative prototype requires FLTK 1.4 or later")
    endif()
    execute_process(COMMAND "${FLTK_CONFIG}" --cxxflags OUTPUT_VARIABLE FLTK_CXXFLAGS
        OUTPUT_STRIP_TRAILING_WHITESPACE)
    execute_process(COMMAND "${FLTK_CONFIG}" --ldflags OUTPUT_VARIABLE FLTK_LDFLAGS
        OUTPUT_STRIP_TRAILING_WHITESPACE)
    separate_arguments(FLTK_CXXFLAGS UNIX_COMMAND "${FLTK_CXXFLAGS}")
    separate_arguments(FLTK_LDFLAGS UNIX_COMMAND "${FLTK_LDFLAGS}")
    if(LAYERC_PROTO_SYSROOT)
        # Keep system paths that exist; add the extracted copy of each.
        foreach(list_name FLTK_CXXFLAGS FLTK_LDFLAGS)
            set(relocated)
            foreach(flag IN LISTS ${list_name})
                if(flag MATCHES "^-([IL])(/.*)$" AND
                   EXISTS "${LAYERC_PROTO_SYSROOT}${CMAKE_MATCH_2}")
                    list(APPEND relocated "-${CMAKE_MATCH_1}${LAYERC_PROTO_SYSROOT}${CMAKE_MATCH_2}")
                endif()
                list(APPEND relocated "${flag}")
            endforeach()
            set(${list_name} ${relocated})
        endforeach()
        list(PREPEND FLTK_LDFLAGS "-L${LAYERC_PROTO_SYSROOT}/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}")
    endif()
    set(FLTK_FOUND TRUE)
endif()

if(GTK4_FOUND)
    add_executable(layere_proto_gtk gtk_proto.cpp)
    target_link_libraries(layere_proto_gtk PRIVATE layerc_editor_core PkgConfig::GTK4)
    target_compile_features(layere_proto_gtk PRIVATE cxx_std_20)
endif()
if(FLTK_FOUND)
    add_executable(layere_proto_fltk fltk_proto.cpp)
    target_compile_options(layere_proto_fltk PRIVATE ${FLTK_CXXFLAGS})
    if(LAYERC_PROTO_SYSROOT)
        target_include_directories(layere_proto_fltk SYSTEM PRIVATE
            "${LAYERC_PROTO_SYSROOT}/usr/include")
    endif()
    target_link_libraries(layere_proto_fltk PRIVATE layerc_editor_core ${FLTK_LDFLAGS})
    target_compile_features(layere_proto_fltk PRIVATE cxx_std_20)
    if(ATK_BRIDGE_FOUND)
        add_executable(layere_proto_fltk_a11y fltk_proto.cpp fltk_a11y.cpp)
        target_compile_definitions(layere_proto_fltk_a11y PRIVATE LAYERC_PROTO_ATSPI
            LAYERC_PROTO_BRIDGE_DIR="${ATK_BRIDGE_CANONICAL_DIR}")
        target_compile_options(layere_proto_fltk_a11y PRIVATE
            ${FLTK_CXXFLAGS} ${ATK_BRIDGE_CFLAGS_OTHER})
        target_include_directories(layere_proto_fltk_a11y SYSTEM PRIVATE ${ATK_BRIDGE_INCLUDES})
        if(LAYERC_PROTO_SYSROOT)
            target_include_directories(layere_proto_fltk_a11y SYSTEM PRIVATE
                "${LAYERC_PROTO_SYSROOT}/usr/include")
        endif()
        # The ATK libraries precede FLTK's search paths, and the run path makes
        # the loader select that same verified adaptor without LD_LIBRARY_PATH.
        target_link_libraries(layere_proto_fltk_a11y PRIVATE
            layerc_editor_core ${ATK_BRIDGE_LINK} ${FLTK_LDFLAGS})
        set_target_properties(layere_proto_fltk_a11y PROPERTIES
            BUILD_RPATH "${ATK_BRIDGE_RUNTIME_DIRS}")
        # DT_RPATH (unlike DT_RUNPATH) also applies to the adaptor's own
        # libatspi dependency, keeping the whole bridge from one install.
        target_link_options(layere_proto_fltk_a11y PRIVATE "LINKER:--disable-new-dtags")
        target_compile_features(layere_proto_fltk_a11y PRIVATE cxx_std_20)
    endif()
endif()
# Kernel-enforced fixture confinement check; links no toolkit.
add_executable(layere_proto_confinement confinement_probe.cpp)
target_link_libraries(layere_proto_confinement PRIVATE layerc_editor_core)
target_compile_features(layere_proto_confinement PRIVATE cxx_std_20)
foreach(target IN ITEMS layere_proto_fltk layere_proto_gtk layere_proto_fltk_a11y layere_proto_confinement)
    if(TARGET ${target})
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
